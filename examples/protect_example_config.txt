# Example integration into the main BruteFIR config on Holographer.
# Channel names match the real output names from all six
# config/sqnt440/sqnt440<rate> configs: "subl"/"subr" (woofers),
# "esll"/"eslr" (ESL panels) -- verified by grep against all six
# configs, not guessed (2026-08-22).
#
# AS OF 2026-08-27, after a REW THD sweep measurement campaign and a
# hardware change: the Neutrik NA2-TX 1:1 isolation transformer was
# REMOVED from the ESL chain and replaced with a passive Neutrik NA2
# FP-MF adapter plug (XLR female -> RCA/phono, wired per IEC 268-12,
# pin 2 = signal, pins 1+3 = ground -- straight wiring, no
# transformer, no saturation behaviour). Reason: no ground loop
# present, and the tonal signature of the old transformer was
# unwanted.
#
# All max_vrms values in the ESL group are now derived from real REW
# THD sweeps (2% THD chosen as the hard ceiling, based on listening:
# 3% already sounds audibly "broken" on this system), not from
# datasheet/bias-voltage estimates. Woofer group: the xmax limit was
# likewise updated from measurement; amp_clip/driver_thermal are
# unchanged (outside the measured range, still datasheet values).
#
# IMPORTANT: this lexer only understands "#" line comments, no /* */
# block comments -- every line must start its own "#".

logic: "protect" {
    output_scale: 2147483647.0;
    # CRITICAL, empirically verified via dry-test on 2026-08-22:
    # BruteFIR's internal buffer at the output_timed() hook is NOT
    # normalised to +-1.0 for integer output formats (S32_LE, as used
    # by every real sqnt440<rate> config) -- it carries the raw
    # integer magnitude. See the project's internal notes
    # ("Software-Trockentest") for the full finding.

    # --- Woofer chain: Sabaj A20a, direct XLR, BTL ---
    chain: "mytek_direct_xlr" {
        dac_max_output_vrms: 6.91;   # Mytek 0dBFS, per factory
                                      # calibration -15dBFS=1.228Vrms
        {
            gain_db: 36.6;
            # Sabaj A20a BTL voltage gain -- MEASURED (2026-08-27):
            # 4.686Vrms at the Sabaj output at -40dBFS (30Hz reference
            # point from the REW measurement campaign), back-derived
            # against the DAC output voltage at that level. Replaces
            # the old PROVISIONAL web-sourced estimate (38.8dB,
            # difference -2.2dB).
        };
    };

    # --- ESL chain: TRI 88 Signature Main In, via a passive Neutrik
    #     NA2 FP-MF adapter plug (XLR->RCA, straight wiring, no
    #     transformer anymore), then the ESL's own 1:60 step-up ---
    chain: "mytek_via_rca_adapter" {
        dac_max_output_vrms: 6.91;
        {
            gain_db: 0.0;
            # Stage 1: Neutrik NA2 FP-MF adapter plug (XLR female ->
            # RCA/phono, wired per IEC 268-12, pin 2 = signal, pins
            # 1+3 = ground). Pure passive wiring, no transformer --
            # no saturation behaviour, assumed loss/gain = 0dB. Correct
            # here if it's ever measured to be otherwise.
            #
            # REPLACES the old Neutrik NA2-TX 1:1 isolation transformer
            # (in the signal path until 2026-08-26) -- that component's
            # own "transformer_saturation" checkpoint is therefore
            # removed entirely (see group below).
        };
        {
            gain_db: 24.5;
            # Stage 2: TRI 88 Signature Main-In gain -- MEASURED
            # (2026-08-27): 8.52Vrms at the TRI output at -22.68dBFS/
            # 80Hz (the 1% THD reference point from the REW measurement
            # campaign), back-derived against the DAC output voltage.
            # Now unambiguously attributable to the TRI alone, since
            # stage 1 (the adapter) has no gain of its own -- previously
            # (with the Neutrik transformer in the path) this figure
            # was an inseparable Neutrik+TRI combination. Replaces the
            # old PROVISIONAL web-sourced estimate (31.0dB).
        };
        {
            gain_db: 35.6;
            # Stage 3: the ESL's own step-up transformer, 1:60 (user-
            # confirmed): 20*log10(60). Unchanged. The transformer is
            # rated for ~100W itself -- its own saturation/thermal
            # behaviour is deliberately not modelled (deferred at the
            # user's request).
        };
    };

    # --- Woofer group ---
    group: {
        channels: "subl", "subr";
        chain: "mytek_direct_xlr";

        limit: {
            name: "xmax";
            max_vrms: 6.68;
            # MEASURED (2026-08-27), REW THD sweep: the 2% THD point at
            # 30Hz = -36.89dBFS -> 6.68Vrms at the Sabaj output (using
            # the newly measured 36.6dB gain). 30Hz was more
            # restrictive than 20Hz in the sweep (2% THD at 20Hz only
            # reached at 12.68Vrms) -- taken as the conservative
            # reference point. Replaces the old theoretical Xmax
            # working value (20.0V, never measured). The root cause of
            # the 30Hz > 20Hz asymmetry is still open (possibly driver
            # nonlinearity or similar) -- not critical for the
            # protection purpose, since the measured value is used
            # directly, with no model assumption in between.
            type: "peak";
            ref_freq_hz: 39.04;
            slope_db_oct: 12.0;
            # Sealed-box Fc -- CORRECTED (2026-08-27): the driver is a
            # Vifa/Scan-Speak PL22WR09-08 (8 ohm), NOT the
            # M21WO4604CUTED previously assumed. Vas=56L (instead of
            # 88.9L) -> Vas_eff=168L (3 drivers sharing one enclosure)
            # instead of 266.7L; with the same 154L net volume:
            # Fc=27*sqrt(1+168/154)=39.04Hz (instead of 44.6Hz),
            # Qtc=0.492 (instead of 0.446, still well under 0.707, no
            # resonance peak). The frequency-weighting structure
            # (12dB/octave) remains valid, only the anchor point moved.
            attack_ms: 1.0;
            release_ms: 200.0;
        };
        limit: {
            name: "amp_clip";
            max_vrms: 52.89;
            # UNCHANGED -- 37.4Vrms (Sabaj rated output @350W/4R BTL)
            # * sqrt(2). Outside the measured range (30Hz/2% THD sits
            # at 6.68V, far below) -- the amp's own rating is
            # demonstrably not the limiting factor, no new measurement
            # needed.
            type: "peak";
            attack_ms: 0.3;
            release_ms: 200.0;
        };
        limit: {
            name: "driver_thermal";
            max_vrms: 37.9;
            # UNCHANGED, BUT TECHNICALLY STALE (2026-08-27): with the
            # correct driver (PL22WR09-08, Pe=125W nominal/350W
            # long-term instead of the old incorrect 80W assumption)
            # this value should properly be higher. Deliberately not
            # recalculated because it's moot: the real 2% THD point
            # (6.68V, see "xmax" above) sits so far below it that the
            # thermal limit never engages either way -- the THD
            # criterion is the actually binding factor, not
            # temperature. Only worth recalculating if the xmax limit
            # is ever relaxed later.
            type: "rms";
            window_ms: 500.0;
            rms_release_ms: 2000.0;
        };
    };

    # --- ESL group: only TWO checkpoints now (transformer_saturation
    #     removed, since the Neutrik transformer is no longer in the
    #     signal path) ---
    group: {
        channels: "esll", "eslr";
        chain: "mytek_via_rca_adapter";

        limit: {
            name: "tri_rated_power";
            max_vrms: 16.19;
            # MEASURED (2026-08-27): the 2% THD point at 80Hz (worst
            # case vs. 130Hz, where it's 25.04V) = -17.1dBFS -> 16.19V
            # at the TRI output, using the newly measured 24.5dB chain
            # gain. Replaces the old datasheet value (33.46V, from the
            # 70W/8R rating) -- the TRI itself still has plenty of
            # power headroom at this level; the THD criterion engages
            # well before the amp's own rating would.
            type: "peak";
            attack_ms: 0.5;
            release_ms: 300.0;
            stage: 2;
            # Stage 1+2 (adapter+TRI) -- measured at the TRI output,
            # before the step-up.
        };
        limit: {
            name: "els_xmax";
            max_vrms: 971.0;
            # MEASURED (2026-08-27), REW THD sweep, full chain: the 2%
            # THD point at 80Hz = -17.1dBFS -> 971V at the stator
            # (using 60.1dB total chain gain: 24.5dB adapter+TRI +
            # 35.6dB own step-up). 80Hz was the most restrictive point
            # across the full sweep (80-300Hz) -- 130Hz already allows
            # 1503V, and 200Hz is even less critical (per the user's
            # direct listening assessment, not separately measured).
            # The behaviour (ceiling RISING with frequency) points to
            # core saturation somewhere in the chain, not a diaphragm
            # excursion limit -- accordingly deliberately NOT
            # frequency-weighted, but set as a flat value at the worst
            # measured point (80Hz): conservative and safe, even though
            # it leaves headroom unused at 130Hz and above.
            #
            # Replaces the old bias-voltage placeholder (2500V,
            # explicitly marked "not measured, pragmatic
            # approximation") -- the real value is only about 39% of
            # that.
            type: "peak";
            attack_ms: 0.5;
            release_ms: 300.0;
            # no "stage" field -> full chain (adapter+TRI+step-up, 60.1dB)
        };
    };
};
